cmake_minimum_required(VERSION 3.16)
project(ivrTTS VERSION 1.0 LANGUAGES CXX)
find_package(CURL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CLIENT_JSON REQUIRED IMPORTED_TARGET json-c)
add_library(ivrTTS ivrTTS.cpp)
target_compile_features(ivrTTS PUBLIC cxx_std_17)
target_include_directories(ivrTTS PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:include>)
target_link_libraries(ivrTTS PUBLIC CURL::libcurl)
target_link_libraries(ivrTTS PRIVATE PkgConfig::CLIENT_JSON)
target_compile_definitions(ivrTTS PRIVATE IVRTTS_DEFAULT_VOICE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/voices")
add_executable(ivrtts_cli main.cpp)
target_link_libraries(ivrtts_cli PRIVATE ivrTTS)
add_executable(exampleTTS exampleTTS.cpp)
target_link_libraries(exampleTTS PRIVATE ivrTTS)
option(IVRTTS_BUILD_TUI "Build terminal player with ncurses and SDL2" ON)
if(IVRTTS_BUILD_TUI)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(TUI_CURSES REQUIRED IMPORTED_TARGET ncursesw)
    pkg_check_modules(TUI_SDL REQUIRED IMPORTED_TARGET sdl2)
    find_package(Threads REQUIRED)
    target_sources(ivrtts_cli PRIVATE tui.cpp)
    target_compile_definitions(ivrtts_cli PRIVATE IVRTTS_HAS_TUI=1)
    target_link_libraries(ivrtts_cli PRIVATE PkgConfig::TUI_CURSES PkgConfig::TUI_SDL Threads::Threads)
endif()
file(COPY voices DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
option(IVRTTS_BUILD_TESTS "Build C++ integration tests" OFF)
if(IVRTTS_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    find_library(TEST_UTIL_LIBRARY util)
    if(NOT TEST_UTIL_LIBRARY)
        message(FATAL_ERROR "C++ tests require libutil for pseudo-terminals")
    endif()
    add_executable(ivrtts_tests test_client.cpp)
    target_compile_definitions(ivrtts_tests PRIVATE TEST_VOICES="${CMAKE_CURRENT_SOURCE_DIR}/voices")
    target_link_libraries(ivrtts_tests PRIVATE ivrTTS PkgConfig::CLIENT_JSON Threads::Threads ${TEST_UTIL_LIBRARY})
    add_test(NAME ivrtts_transport COMMAND ivrtts_tests cli $<TARGET_FILE:ivrtts_cli>)
    if(IVRTTS_BUILD_TUI)
        add_test(NAME ivrtts_tui COMMAND ivrtts_tests tui $<TARGET_FILE:ivrtts_cli>)
    endif()
    set_tests_properties(ivrtts_transport PROPERTIES TIMEOUT 45)
    if(IVRTTS_BUILD_TUI)
        set_tests_properties(ivrtts_tui PROPERTIES TIMEOUT 45)
    endif()
endif()
include(GNUInstallDirs)
install(TARGETS ivrTTS ivrtts_cli
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ivrTTS.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY voices/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ivrTTS/voices)
